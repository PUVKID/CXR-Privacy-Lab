+def r_precision(

+def shift_indices_tuple(indices_tuple, batch_size):
+    """
+    Shifts indices of positives and negatives of pairs or triplets by batch_size
+
+    if len(indices_tuple) != 3 or len(indices_tuple) != 4, it will return indices_tuple
+    Args:
+        indices_tuple is a tuple with torch.Tensor
+        batch_size is an int
+    Returns:
+        A tuple with shifted indices
+    """
+
+    if len(indices_tuple) == 3:
+        indices_tuple = (indices_tuple[0],) + tuple(
+            [x + batch_size if len(x) > 0 else x for x in indices_tuple[1:]]
+        )
+    elif len(indices_tuple) == 4:
+        indices_tuple = tuple(
+            [
+                x + batch_size if len(x) > 0 and i % 2 == 1 else x
+                for i, x in enumerate(indices_tuple)
+            ]
+        )
+    return indices_tuple

+def is_distributed():

+# modified from https://github.com/JohnGiorgi/DeCLUTR
+def all_gather(embeddings, labels):
+    labels = labels.to(embeddings.device)
+    # If we are not using distributed training, this is a no-op.
+    if not is_distributed():
+        return embeddings, labels
+    world_size = torch.distributed.get_world_size()
+    rank = torch.distributed.get_rank()
+    # Gather the embeddings on all replicas
+    embeddings_list = [torch.ones_like(embeddings) for _ in range(world_size)]
+    labels_list = [torch.ones_like(labels) for _ in range(world_size)]
+    torch.distributed.all_gather(embeddings_list, embeddings.contiguous())
+    torch.distributed.all_gather(labels_list, labels.contiguous())
+    # The gathered copy of the current replicas embeddings have no gradients, so we overwrite
+    # them with the embeddings generated on this replica, which DO have gradients.
+    embeddings_list[rank] = embeddings
+    labels_list[rank] = labels
+    # Finally, we concatenate the embeddings
+    embeddings = torch.cat(embeddings_list)
+    labels = torch.cat(labels_list)
+    return embeddings, labels
+
+
+def all_gather_embeddings_labels(embeddings, labels):
+    if c_f.is_list_or_tuple(embeddings):
+        assert c_f.is_list_or_tuple(labels)
+        all_embeddings, all_labels = [], []
+        for i in range(len(embeddings)):
+            E, L = all_gather(embeddings[i], labels[i])
+            all_embeddings.append(E)
+            all_labels.append(L)
+        embeddings = torch.cat(all_embeddings, dim=0)
+        labels = torch.cat(all_labels, dim=0)
+    else:
+        embeddings, labels = all_gather(embeddings, labels)
+
+    return embeddings, labels

+class MatchFinder:
+    def __init__(self, distance, threshold=None):
+        self.distance = distance
+        self.threshold = threshold
+
+    def operate_on_emb(self, input_func, query_emb, ref_emb=None, *args, **kwargs):
+        if ref_emb is None:
+            ref_emb = query_emb
+        return input_func(query_emb, ref_emb, *args, **kwargs)
+
+    # for a batch of queries
+    def get_matching_pairs(
+        self, query_emb, ref_emb=None, threshold=None, return_tuples=False
+    ):
+        with torch.no_grad():
+            threshold = threshold if self.threshold is None else self.threshold
+            return self.operate_on_emb(
+                self._get_matching_pairs, query_emb, ref_emb, threshold, return_tuples
+            )
+
+    def _get_matching_pairs(self, query_emb, ref_emb, threshold, return_tuples):
+        mat = self.distance(query_emb, ref_emb)
+        matches = mat >= threshold if self.distance.is_inverted else mat <= threshold
+        matches = matches.cpu().numpy()
+        if return_tuples:
+            return list(zip(*np.where(matches)))
+        return matches
+
+    # where x and y are already matched pairs
+    def is_match(self, x, y, threshold=None):
+        threshold = threshold if self.threshold is None else self.threshold
+        with torch.no_grad():
+            dist = self.distance.pairwise_distance(x, y)
+            output = (
+                dist >= threshold if self.distance.is_inverted else dist <= threshold
+            )
+            if output.nelement() == 1:
+                return output.detach().item()
+            return output.cpu().numpy()

+def mining_order(rows, *, seed, block_size=8):
+    """공식 build_mining_list_32/get_block의 블록8 규칙을 이식합니다."""
+    if block_size != 8:

+    names = np.arange(len(rows))
+    labels = np.asarray([int(row[1]) for row in rows], dtype=np.int64)
+    if any(value < 2 for value in Counter(labels).values()):
+        raise ValueError("검색 학습에는 원본 train.csv의 다중 영상 환자만 허용합니다.")
+    rng = np.random.RandomState(seed)
+    coin = random.Random(seed)
+    output, remainder = [], []
+
+    def take(current, count):
+        nonlocal names, labels
+        chosen = rng.choice(names[labels == current], int(count), replace=False)
+        keep = ~np.isin(names, chosen)
+        names, labels = names[keep], labels[keep]
+        return chosen.tolist()
+
+    while len(names):
+        current = rng.choice(labels)
+        count = int(np.sum(labels == current))
+        block = []
+        success = True
+        if count % block_size == 0:
+            block += take(current, block_size)
+        else:
+            if count > block_size:
+                if count % 8 == 1:
+                    block += take(current, 2)
+                elif coin.random() < .5:
+                    block += take(current, block_size)
+                elif count % block_size == block_size - 1:
+                    block += take(current, count % block_size - 2)
+                else:
+                    block += take(current, count % block_size)
+            else:
+                block += take(current, count % block_size)
+            rest = block_size - len(block)
+            while rest:
+                ids, counts = np.unique(labels, return_counts=True)
+                mod = counts % block_size
+                possible = ids[(mod > 1) & (((rest - mod) > 1) | ((rest - mod) == 0))]
+                if not len(possible):
+                    possible = ids[mod == block_size - 1]
+                    if not len(possible):
+                        success = False
+                    else:
+                        block += take(rng.choice(possible), rest)
+                    break
+                current = rng.choice(possible)
+                block += take(current, int(mod[ids == current].item()))

+        (output if success else remainder).extend(block)
+    result = output + remainder
+    if len(result) != len(rows) or len(set(result)) != len(rows):

+def optimizer_batch(model, optimizer, scaler, forward, *, memory=None, on_overflow=None):
+    """One original native-AMP batch; overflow skips its optimizer update.
+
+The previous port replayed an overflowing batch and rolled back the CBM/BN.
+Native GradScaler used by the author's Lightning trainer skips the optimizer
+step, lowers the scale, and proceeds. Keep that behavior and expose it in logs.
+"""
+    start_scale = scaler.get_scale()
+    optimizer.zero_grad(set_to_none=True)
+    loss = forward()
+    if not torch.isfinite(loss):
+        raise FloatingPointError("학습 손실이 NaN/무한대입니다. 계속 학습하지 않습니다.")
+    scaler.scale(loss).backward()
+    scaler.unscale_(optimizer)
+    gradients = [(name, parameter.grad) for name, parameter in model.named_parameters()
+                 if parameter.grad is not None]
+    checks = [torch.isfinite(value).all() for _, value in gradients]
+    finite = bool(torch.stack(checks).all()) if checks else False
+    first_bad = next((name for (name, _), good in zip(gradients, checks) if not good), None)
+    if not gradients or (not finite and not scaler.is_enabled()):
+        raise FloatingPointError(f"유한 기울기를 얻지 못했습니다. 최초 위치: {first_bad}")
+    scaler.step(optimizer)
+    scaler.update()
+    result = {"gradients_finite": finite, "amp_overflows": int(not finite),
+              "optimizer_applied": finite,
+              "scale_before": start_scale, "scale_after": scaler.get_scale()}
+    if not finite:
+        if not math.isfinite(scaler.get_scale()) or scaler.get_scale() <= 0:
+            raise FloatingPointError("AMP scale became invalid")
+        if on_overflow is not None:
+            on_overflow({**result, "overflow_attempt": 1,
+                         "first_nonfinite_parameter": first_bad})
+    return loss.detach(), result

+        super().__init__()

+        self.register_buffer("labels", torch.zeros(memory_size, dtype=torch.long))

+        self.register_buffer("count", torch.zeros((), dtype=torch.long))

+        self.kernel.label_memory = self.labels.to(labels.device, labels.dtype)
